name: Build APK

on:
  push:
    branches: [ main ]
  workflow_dispatch:

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Setup Node
        uses: actions/setup-node@v4
        with:
          node-version: 20

      - name: Setup Java
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 21

      - name: Install Capacitor
        run: |
          npm init -y
          npm install @capacitor/core@6 @capacitor/cli@6 @capacitor/android@6

      - name: Prepare project
        run: |
          mkdir -p www
          HTMLFILE=$(ls *.html | head -n 1)
          cp "$HTMLFILE" www/index.html
          npx cap init "My Game" com.saeid.mygame --web-dir www
          npx cap add android

      - name: Build APK
        run: |
          cd android
          chmod +x gradlew
          ./gradlew assembleDebug

      - name: Upload APK
        uses: actions/upload-artifact@v4
        with:
          name: game-apk
          path: android/app/build/outputs/apk/debug/app-debug.apk
